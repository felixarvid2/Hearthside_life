import { readFileSync } from 'node:fs';
import { describe, it } from 'vitest';

// Temporary inspection helper (deleted after task H grip design).
describe('tmp bone inspect', () => {
  it('prints hand nodes', () => {
    const buf = readFileSync('public/assets/characters/mpfb_m.glb');
    const jsonLen = buf.readUInt32LE(12);
    const json = JSON.parse(buf.subarray(20, 20 + jsonLen).toString('utf8')) as { nodes: Array<Record<string, unknown>> };
    const nodes = json.nodes;
    const byName = new Map<string, number>();
    nodes.forEach((n, i) => { const nm = n.name; if (typeof nm === 'string') byName.set(nm, i); });
    const parent = new Map<number, number>();
    nodes.forEach((n, i) => { for (const c of (n.children as number[] | undefined) ?? []) parent.set(c, i); });
    for (const name of ['root', 'pelvis', 'clavicle_r', 'upperarm_r', 'lowerarm_r', 'hand_r', 'middle_01_r', 'thumb_01_r', 'index_01_r', 'hand_l', 'middle_01_l']) {
      const i = byName.get(name);
      if (i === undefined) { console.log(name, 'MISSING'); continue; }
      const n = nodes[i]!;
      console.log(name, 'parent', nodes[parent.get(i)!]?.name, 'T', n.translation, 'R', n.rotation, 'S', n.scale);
    }
    // model-space position via accumulated translation*rotation (no scale) up the chain
    const pos = new Map<string, [number, number, number]>();
    const walk = (i: number, p: [number, number, number], q: [number, number, number, number]): void => {
      // minimal quaternion rotate
      const n = nodes[i]!;
      const t = (n.translation as number[] | undefined) ?? [0, 0, 0];
      const r = (n.rotation as number[] | undefined) ?? [0, 0, 0, 1];
      const [px, py, pz] = p;
      const [qx, qy, qz, qw] = q;
      const [rx, ry, rz, rw] = r;
      // rotate t by q
      const ix = qw * t[0]! + qy * t[2]! - qz * t[1]!;
      const iy = qw * t[1]! + qz * t[0]! - qx * t[2]!;
      const iz = qw * t[2]! + qx * t[1]! - qy * t[0]!;
      const iw = -qx * t[0]! - qy * t[1]! - qz * t[2]!;
      const wx = ix * qw + iw * -qx + iy * -qz - iz * -qy;
      const wy = iy * qw + iw * -qy + iz * -qx - ix * -qz;
      const wz = iz * qw + iw * -qz + ix * -qy - iy * -qx;
      const np: [number, number, number] = [px + wx, py + wy, pz + wz];
      const nm = n.name as string | undefined;
      if (nm) pos.set(nm, np);
      // q' = q * r
      const nqx = qw * rx + qx * rw + qy * rz - qz * ry;
      const nqy = qw * ry + qy * rw + qz * rx - qx * rz;
      const nqz = qw * rz + qz * rw + qx * ry - qy * rx;
      const nqw = qw * rw - qx * rx - qy * ry - qz * rz;
      for (const c of (n.children as number[] | undefined) ?? []) walk(c, np, [nqx, nqy, nqz, nqw]);
    };
    const rootIdx = byName.get('root');
    if (rootIdx !== undefined) walk(rootIdx, [0, 0, 0], [0, 0, 0, 1]);
    for (const name of ['hand_r', 'middle_01_r', 'hand_l', 'middle_01_l']) console.log(name, 'model', pos.get(name));
  });
});
